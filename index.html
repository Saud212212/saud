import express from 'express'; import cors from 'cors'; import helmet from 'helmet'; import { PrismaClient } from '@prisma/client'; import { clerkMiddleware, requireAuth } from '@clerk/express'; import Stripe from 'stripe'; const app = express(); const prisma = new PrismaClient(); const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!); app.use(helmet()); app.use(cors({ origin: process.env.FRONTEND_URL })); app.use(express.json()); app.use(clerkMiddleware()); // ============ HEALTH CHECK ============ app.get('/health', (req, res) => res.json({ ok: true })); // ============ USER PROFILE ============ app.get('/api/me', requireAuth(), async (req, res) => { const user = await prisma.user.findUnique({ where: { clerkId: req.auth.userId }, include: { subscription: true } }); res.json(user); }); // ============ SAVE CALCULATION ============ app.post('/api/calculations', requireAuth(), async (req, res) => { const { calculatorId, inputs, results, title } = req.body; const calc = await prisma.calculation.create({ data: { userId: req.auth.userId, calculatorId, inputs, results, title } }); res.json(calc); }); // ============ CREATE STRIPE CHECKOUT ============ app.post('/api/checkout', requireAuth(), async (req, res) => { const { priceId } = req.body; const session = await stripe.checkout.sessions.create({ mode: 'subscription', line_items: [{ price: priceId, quantity: 1 }], success_url: `${process.env.FRONTEND_URL}/success`, cancel_url: `${process.env.FRONTEND_URL}/pricing`, customer_email: req.auth.email }); res.json({ url: session.url }); }); // ============ STRIPE WEBHOOK ============ app.post('/api/webhook/stripe', express.raw({ type: 'application/json' }), async (req, res) => { const event = stripe.webhooks.constructEvent( req.body, req.headers['stripe-signature']!, process.env.STRIPE_WEBHOOK_SECRET! ); if (event.type === 'customer.subscription.created') { // Activate subscription in DB } res.json({ received: true }); } ); app.listen(3001, () => console.log('API ready :3001'));

ربط الدومين بـ Cloudflare DNS 10 دقائق

Cloudflare يعطيك: CDN عالمي + حماية DDoS + SSL تلقائي + سرعة أعلى. كل ذلك مجاناً.

الخطوات:
  • أنشئ حساباً مجانياً على cloudflare.com
  • اضغط "Add Site" وأدخل دومينك
  • Cloudflare يفحص سجلات DNS الحالية تلقائياً
  • اختر الخطة المجانية (تكفي جداً)
  • غيّر nameservers من لوحة الـ Registrar الأصلي للتي يعطيك إياها Cloudflare
  • انتظر 5 دقائق - 24 ساعة (عادة دقائق)

SSL/TLS تلقائي فوري

تشفير https:// المتصفح يجعله إلزامياً لأي موقع تجاري.

على Cloudflare: اذهب لـ SSL/TLS → Overview واختر "Full (Strict)". هذا يفعّل SSL تلقائياً.

على Vercel: SSL مفعّل افتراضياً لكل دومين. لا تحتاج إعداد.

تحقق: افتح https://www.ssllabs.com/ssltest/ وأدخل دومينك. يجب أن تحصل على درجة A أو A+.
